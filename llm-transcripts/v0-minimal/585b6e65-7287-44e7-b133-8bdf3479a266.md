typedef struct {
    /* The universal fallback. JSON-in, JSON-out. Every spec must
     * implement this — it's the cross-language lingua franca. */
    int (*invoke)(handle, fn, in_json,  in_size,  out_json,  out_cap, out_size);

    /* The fast path. The spec defines what "native" means for
     * itself — a Lua spec might pass values via the worker's
     * lua_State, a C spec might pass typed structs, a future
     * msgpack-aware spec might pass packed binary. Optional;
     * NULL means "no fast path available, always go through invoke". */
    int (*invoke_native)(handle, fn, native_in, in_size, native_out, out_cap, out_size);

    /* The bridge. Used at language boundaries: a Lua producer
     * feeding a C consumer goes Lua-native → JSON → C-native. */
    int (*native_to_json)(...);
    int (*json_to_native)(...);
} lang_spec_t;
