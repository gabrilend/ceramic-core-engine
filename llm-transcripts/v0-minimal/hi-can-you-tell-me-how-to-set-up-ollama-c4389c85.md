```
#!/bin/bash
# ollama-build-and-deploy.sh
#
# Build a fully static, musl-linked ollama binary on Alpine, then ship the
# source tree and the compiled binary to two peer hosts. If the most recent
# upstream release tag already matches the version we have on disk, the build
# step is skipped — but the deploy step still runs (idempotent at the peer).
#
# Why this shape: we want one build host doing the heavy lifting, and the
# peers receiving both the binary AND the source so any of them can rebuild
# without depending on this script being re-run. Hardware is identical across
# the three hosts, so the binary is portable.

set -euo pipefail

# {{{ config — override DIR via $1 if you want a different install root
DIR="${1:-/home/sora/programs/ollama}"

PEER_1="192.168.1.12"
PEER_2="192.168.1.13"
PEER_USER="sora"

SRC_DIR="${DIR}/src"
BIN_DIR="${DIR}/bin"
DIST_DIR="${DIR}/dist"
LOG_DIR="${DIR}/tmp"

VERSION_FILE="${DIR}/.built-version"
OLLAMA_REPO="https://github.com/ollama/ollama.git"
GITHUB_API="https://api.github.com/repos/ollama/ollama/releases/latest"
# }}}

# {{{ setup_dirs()
# Create the on-disk layout. The log directory is symlinked into /tmp so build
# noise stays in RAM rather than wearing the disk.
setup_dirs() {
    mkdir -p "${DIR}"
    mkdir -p "${BIN_DIR}"
    mkdir -p "${DIST_DIR}"
    if [ ! -L "${LOG_DIR}" ]; then
        local ram_log="/tmp/ollama-build-$(id -u)"
        mkdir -p "${ram_log}"
        ln -snf "${ram_log}" "${LOG_DIR}"
    fi
}
# }}}

# {{{ ensure_build_deps()
# build-base = gcc+g+++make; cmake for llama.cpp; git+bash+curl for plumbing.
# go must already be installed via apk — if it isn't, we stop here so the user
# can fix the community-repo situation rather than have the script silently
# fall back to a glibc Go shim.
ensure_build_deps() {
    apk add --no-cache build-base cmake git bash curl
    if ! command -v go >/dev/null 2>&1; then
        echo "ERROR: 'go' is not installed." >&2
        echo "Try: apk update && apk add go" >&2
        echo "If apk doesn't find it, enable the community repo for your" >&2
        echo "Alpine version in /etc/apk/repositories and apk update." >&2
        exit 1
    fi
}
# }}}

# {{{ get_latest_tag()
# Ask GitHub for the most recent published release tag. We don't trust git
# ls-remote here because it would include pre-release and draft tags.
get_latest_tag() {
    local response
    response="$(curl -fsSL "${GITHUB_API}")"
    local tag
    tag="$(echo "${response}" | sed -n 's/.*"tag_name": *"\([^"]*\)".*/\1/p' | head -n1)"
    if [ -z "${tag}" ]; then
        echo "ERROR: could not parse latest tag from GitHub API" >&2
        exit 1
    fi
    echo "${tag}"
}
# }}}

# {{{ already_built()
# True if .built-version matches the desired tag AND the binary still exists.
# Either condition failing forces a rebuild — we don't want a stale tag file
# to mask a missing binary.
already_built() {
    local target_tag="$1"
    if [ ! -f "${VERSION_FILE}" ]; then
        return 1
    fi
    if [ ! -x "${BIN_DIR}/ollama" ]; then
        return 1
    fi
    local built_tag
    built_tag="$(cat "${VERSION_FILE}")"
    [ "${built_tag}" = "${target_tag}" ]
}
# }}}

# {{{ fetch_source()
# Shallow clone at the exact tag. Previous checkout is wiped so we never carry
# half-built artifacts from a prior failed run into a new attempt.
fetch_source() {
    local tag="$1"
    rm -rf "${SRC_DIR}"
    git clone --depth=1 --branch "${tag}" "${OLLAMA_REPO}" "${SRC_DIR}"
}
# }}}

# {{{ build_static()
# Two-step build:
#   1. `make` invokes CMake for llama.cpp (produces static .a archives) and
#      then go build for the ollama binary. We pre-set CGO_LDFLAGS so the C++
#      runtime gets pulled in statically.
#   2. We rebuild the Go binary explicitly with -linkmode external and
#      -extldflags '-static', because the upstream Makefile's go invocation
#      doesn't pass those through.
# Final `ldd` check is non-negotiable: if it isn't fully static, we bail loudly
# rather than ship a binary that needs runtime libs the peers might lack.
build_static() {
    cd "${SRC_DIR}"

    export CGO_ENABLED=1
    export CC=gcc
    export CXX=g++
    export CGO_CFLAGS="-O2"
    export CGO_CXXFLAGS="-O2"
    export CGO_LDFLAGS="-static-libstdc++ -static-libgcc"

    local build_log="${LOG_DIR}/build.log"
    make -j"$(nproc)" >"${build_log}" 2>&1

    go build \
        -tags 'netgo osusergo' \
        -ldflags '-linkmode external -extldflags "-static"' \
        -o "${BIN_DIR}/ollama" \
        .

    local ldd_out
    ldd_out="$(ldd "${BIN_DIR}/ollama" 2>&1 || true)"
    if echo "${ldd_out}" | grep -q 'not a dynamic'; then
        echo "OK: binary is fully static."
    else
        echo "ERROR: binary still has dynamic dependencies:" >&2
        echo "${ldd_out}" >&2
        exit 1
    fi
}
# }}}

# {{{ package_dist()
# Bundle the source tree AND the compiled binary into one tarball, so peers
# receive everything needed to either run as-is or rebuild from source.
package_dist() {
    local tag="$1"
    local archive_name="ollama-${tag}.tar.gz"
    local archive_path="${DIST_DIR}/${archive_name}"
    rm -f "${archive_path}"
    tar -czf "${archive_path}" \
        -C "${DIR}" \
        --transform "s,^src,ollama-${tag}/src," \
        --transform "s,^bin,ollama-${tag}/bin," \
        src bin
    echo "Packaged: ${archive_path}"
}
# }}}

# {{{ peer_already_has_version()
# Cheap remote check so we don't re-scp a multi-hundred-MB tarball when the
# peer is already current. If ssh fails, the peer is treated as out-of-date.
peer_already_has_version() {
    local peer="$1"
    local target_tag="$2"
    local peer_version
    peer_version="$(ssh -o ConnectTimeout=5 "${PEER_USER}@${peer}" \
        "cat ${DIR}/.built-version 2>/dev/null || echo none" 2>/dev/null || echo none)"
    [ "${peer_version}" = "${target_tag}" ]
}
# }}}

# {{{ deploy_to_peer()
# Copies tarball to the peer, extracts in place, replaces src/ and bin/,
# records the version. Peer runtime config (serve.sh, openrc unit) is NOT
# touched here — that's a separate one-time provisioning concern.
deploy_to_peer() {
    local peer="$1"
    local tag="$2"
    local archive_name="ollama-${tag}.tar.gz"
    local archive_path="${DIST_DIR}/${archive_name}"

    if peer_already_has_version "${peer}" "${tag}"; then
        echo "Peer ${peer} already at ${tag}, skipping."
        return 0
    fi

    echo "Deploying ${tag} to ${peer}..."
    ssh "${PEER_USER}@${peer}" "mkdir -p ${DIR}"
    scp "${archive_path}" "${PEER_USER}@${peer}:${DIR}/"
    ssh "${PEER_USER}@${peer}" "
        set -e
        cd ${DIR}
        tar -xzf ${archive_name}
        rm -rf src bin
        mv ollama-${tag}/src .
        mv ollama-${tag}/bin .
        rmdir ollama-${tag}
        echo ${tag} > .built-version
        rm -f ${archive_name}
    "
    echo "Peer ${peer} updated to ${tag}."
}
# }}}

# {{{ main
main() {
    setup_dirs
    ensure_build_deps

    local tag
    tag="$(get_latest_tag)"
    echo "Latest upstream release: ${tag}"

    if already_built "${tag}"; then
        echo "Local build already at ${tag}, skipping rebuild."
    else
        fetch_source "${tag}"
        build_static
        echo "${tag}" > "${VERSION_FILE}"
        echo "Local build complete: ${tag}"
    fi

    package_dist "${tag}"
    deploy_to_peer "${PEER_1}" "${tag}"
    deploy_to_peer "${PEER_2}" "${tag}"

    echo "All hosts at ${tag}."
}
# }}}

main "$@"
```
